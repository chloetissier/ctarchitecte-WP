function ct_liste($b, $id) {
    if ($b['blockName'] === 'core/button') {
        echo "page $id bouton : " . trim(wp_strip_all_tags($b['innerHTML'])) . " | url=" . ($b['attrs']['url'] ?? '(aucune)') . "\n";
    }
    foreach ($b['innerBlocks'] as $ib) { ct_liste($ib, $id); }
}

function ct_relier(&$b, $map, &$n) {
    if ($b['blockName'] === 'core/button' && empty($b['attrs']['url'])) {
        $texte = trim(html_entity_decode(wp_strip_all_tags($b['innerHTML']), ENT_QUOTES, 'UTF-8'));
        $texte = str_replace(['’', "\u{00A0}"], ["'", ' '], $texte);
        if (isset($map[$texte])) {
            $url = $map[$texte];
            $b['attrs']['url'] = $url;
            $fix = function ($h) use ($url) { return preg_replace('#<a (class="wp-block-button__link)#', '<a href="' . esc_url($url) . '" $1', $h, 1); };
            $b['innerHTML'] = $fix($b['innerHTML']);
            foreach ($b['innerContent'] as $i => $c) { if (is_string($c)) { $b['innerContent'][$i] = $fix($c); } }
            $n++;
            echo "  relié : $texte -> $url\n";
        }
    }
    foreach ($b['innerBlocks'] as &$ib) { ct_relier($ib, $map, $n); }
}

// Transforme une page Kadence : fonds blancs, sans image de fond (hero), coquilles corrigées
function ct_blanchir($content) {
    $blocks = parse_blocks($content);
    $f = function (&$b) use (&$f) {
        if ($b['blockName'] === 'kadence/rowlayout') {
            foreach (['bgImg', 'bgImgID', 'bgImgPosition', 'overlayGradient', 'currentOverlayTab', 'overlayOpacity'] as $k) { unset($b['attrs'][$k]); }
            $b['attrs']['bgColor'] = '';
            if (!empty($b['attrs']['mobileBackground'][0])) { $b['attrs']['mobileBackground'][0]['enable'] = false; $b['attrs']['mobileBackground'][0]['bgImg'] = ''; $b['attrs']['mobileBackground'][0]['bgImgID'] = ''; }
            if (!empty($b['attrs']['mobileOverlay'][0])) { $b['attrs']['mobileOverlay'][0]['enable'] = false; }
        }
        foreach ($b['innerBlocks'] as &$ib) { $f($ib); }
    };
    foreach ($blocks as &$b) { $f($b); }
    unset($b);
    $c = serialize_blocks($blocks);
    $corr = ['Budget maitrisé' => 'Budget maîtrisé', 'Rendez – vous' => 'Rendez-vous', 'rendez – vous' => 'rendez-vous', 'Rendez - vous' => 'Rendez-vous', 'rendez - vous' => 'rendez-vous',
             'levée de réserve' => 'levée des réserves', 'Vous savez  où' => 'Vous savez où', 'rangements et matériaux et coloris' => 'rangements, matériaux et coloris',
             'réponse rapide par mail' => 'réponse rapide par e-mail'];
    return str_replace(array_keys($corr), array_values($corr), $c);
}

// Méthode : titre pleine largeur (1 colonne, 30 px), bloc « deux publics » avec grandes photos, nouvel encadré final
function ct_methode($content) {
    $dir = dirname(__DIR__) . '/legal/';
    $publics = parse_blocks(file_get_contents($dir . 'methode-publics.html'));
    $cta = parse_blocks(file_get_contents($dir . 'cta-methode.html'));
    $out = []; $premiere = true;
    foreach (parse_blocks($content) as $b) {
        if ($b['blockName'] === 'kadence/rowlayout') {
            $txt = serialize_block($b);
            if ($premiere) {
                $premiere = false;
                // Retire la 2e colonne vide (place de l'ancienne photo) et passe la rangée en 1 colonne
                if (count($b['innerBlocks']) === 2 && empty($b['innerBlocks'][1]['innerBlocks'])) {
                    array_pop($b['innerBlocks']);
                    $nulls = array_keys($b['innerContent'], null, true);
                    array_splice($b['innerContent'], end($nulls), 1);
                    $b['attrs']['columns'] = 1;
                    echo "1re rangée : 2e colonne vide retirée\n";
                }
            } elseif (strpos($txt, '"id":174') !== false) { foreach ($publics as $p) { if ($p['blockName']) { $out[] = $p; } } echo "bloc deux publics remplacé\n"; continue; }
            elseif (strpos($txt, 'Découvrez mes réalisations') !== false) { foreach ($cta as $p) { if ($p['blockName']) { $out[] = $p; } } echo "boutons du bas remplacés par l'encadré\n"; continue; }
        }
        $out[] = $b;
    }
    $c = serialize_blocks($out);
    return str_replace(['"fontSize":"36px"', 'font-size:36px'], ['"fontSize":"30px"', 'font-size:30px'], $c);
}
